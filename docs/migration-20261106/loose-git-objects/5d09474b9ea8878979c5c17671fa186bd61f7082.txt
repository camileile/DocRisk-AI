  const input = useRef<HTMLInputElement>(null);

  function choose(selected: File | null) {
    setResult(null);
    setError(null);
    if (selected && selected.size > MAX_MB * 1024 * 1024) {
      setError(`O arquivo tem ${(selected.size / 1024 / 1024).toFixed(1)} MB; o limite é ${MAX_MB} MB.`);
      setFile(null);
      return;
    }
    setFile(selected);
  }

  async function send() {
    if (!file || !contract) return;
    setSending(true);
    setError(null);
    const form = new FormData();
    form.append("contract_id", contract);
    form.append("file", file);
    try {
      setResult(await api<{ batch_id: number }>("/api/supplier/batches", { method: "POST", body: form }));
      setFile(null);
      if (input.current) input.current.value = "";
    } catch (err) {
      setError(err instanceof Error ? err.message : "Falha no envio");
    } finally {
      setSending(false);
    }
  }

  return (
    <>
      <PageIntro
        eyebrow="ENVIO DE LOTE"
        title="Um arquivo, vários documentos."
        description="Envie um PDF, imagem ou TIFF multipágina com documentos misturados. O sistema separa, classifica e lê cada página."
      />
      <div className="upload-layout">
        <section className="workspace-card">
          <p className="muted">
            Destino: <strong>{contracts.find((c) => c.id === contract)?.nome ?? "selecione um contrato"}</strong> (altere no
            seletor de contrato acima).
          </p>
          <label
            className="upload-area"
            onDragOver={(e) => e.preventDefault()}
            onDrop={(e) => {
              e.preventDefault();
              choose(e.dataTransfer.files[0] ?? null);
            }}
          >
            <UploadCloud size={34} />
            <strong>{file ? file.name : "Arraste o arquivo ou clique para escolher"}</strong>
            <span>{file ? `${(file.size / 1024).toFixed(0)} KB` : `PDF, PNG, JPG ou TIFF · até ${MAX_MB} MB · um arquivo por lote`}</span>
            <input
              ref={input}
              type="file"
              accept={ACCEPT}
              className="sr-only"
              onChange={(e) => choose(e.target.files?.[0] ?? null)}
            />
          </label>
          {error && <Notice tone="error">{error}</Notice>}
          {result && (
            <Notice tone="success">
              Lote {result.batch_id} recebido e enviado para processamento.{" "}
              <Link href={`/painel/lotes/detalhe?id=${result.batch_id}`}>
                Acompanhar processamento <ArrowUpRight size={14} />
              </Link>
            </Notice>
          )}
          <button className="button" disabled={!file || !contract || sending} onClick={send}>
            <FileUp size={17} /> {sending ? "Enviando…" : "Enviar lote"}
          </button>
        </section>
        <aside className="workspace-card upload-guidance">
          <h2>O que acontece depois</h2>
          <ol>
            <li>O arquivo é validado (formato, integridade, tamanho e número de páginas).</li>
            <li>Cada página é renderizada e lida pela IA (texto, tipo, trabalhador, competência).</li>
            <li>Páginas são agrupadas em documentos pela numeração e pelo conteúdo.</li>
            <li>Erros objetivos (ilegível, página faltante) voltam como pendência com prazo.</li>
            <li>Incertezas vão para a revisão humana da Bernhoeft.</li>
          </ol>
          <p className="card-footnote">Use somente documentos fictícios nesta demonstração.</p>
        </aside>
      </div>
    </>
  );
